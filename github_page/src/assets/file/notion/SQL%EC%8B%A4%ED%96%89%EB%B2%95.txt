4. table 만들기  (varchar() = 문자열 (자바에서 String))

desc test;
=========================
■ 데이터 입력 : insert
■ 데이터 검색 : select
■ 데이터 수정 : update
■ 데이터 삭제 : delete


■ 데이터 입력 : insert
insert into test values('kim','k123','김정은');
insert into test values('lee','l123','이효리');
insert into test values('park','p123','박문수');

■ 데이터 검색 : select

select id, name from test;
select name from test;
select * from test;(모두 본다)

-order by 키필드 asc(오름차순,순서대로) 또는 desc(내림차순, 역순으로)
select * from test order by id asc; ##asc는 생략 가능
select * from test order by id desc;
-'박문수' 검색하시오-
select * from test where name='박문수'

-'이'가 포함된 자료를 검색하세요

select * from test where name like '%이%';

■ 데이터 수정 : update

update test set name = '이순신' where name = '이상순';

■ 데이터 삭제 : delete
'이순신'을 삭제하시오

delete from test where name = '이순신'

■ DataBase 삭제
drop database mydb;

create table board(
 num int not null auto_increment primary key,
 writer varchar(30) not null,
 subject varchar(50) not null
 wdate date
);
