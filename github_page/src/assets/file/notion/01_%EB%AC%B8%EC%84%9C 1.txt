	1. if문
		=====================================================
		- if문 형식
		=====================================================		
				if(조건) {
							조건이 참이면 처리할 문장;
							조건이 참이면 처리할 문장;
						}
		=====================================================
		- if문 형식 2
		=====================================================
				if(조건) {
							조건이 참이면 처리할 문장;
						}
				else	{
							조건이 거짓일 때 처리할 문장;
						}
		=====================================================
		- if문 형식 3
		=====================================================
				if(조건1) {
								조건 1을 처리할 문장;
						 }
				else if(조건2){
									조건 2을 처리할 문장;
							 }				
				else if(조건3){
									조건 3을 처리할 문장;
							 }
				else {
						위의 조건이 모두 false일 때 처리할 문장
					 }
		=====================================================
		아래 예시 (else를 쓰는 것이 아닌 if로 계속 조건을 제시할 시), 사용량이 많아져 비효율적(낭비)
		if(조건){
		}
		if(조건){
		}
		if(조건){
		}
		if(조건){
		}
		=====================================================
